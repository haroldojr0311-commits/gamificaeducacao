const SOBRE = {'0':'⁰','1':'¹','2':'²','3':'³','4':'⁴','5':'⁵','6':'⁶','7':'⁷','8':'⁸','9':'⁹','-':'⁻','+':'⁺','n':'ⁿ','x':'ˣ'};
function sobrescrito(t){ return /^[-+0-9nx]+$/.test(t) ? t.split('').map(c => SOBRE[c]).join('') : '^(' + t + ')'; }
function texSimples(s){
  return String(s)
    .replace(/\\\(|\\\)/g, '')
    .replace(/\\frac\{([^{}]*)\}\{([^{}]*)\}/g, '$1/$2')
    .replace(/\\sqrt\{([^{}]*)\}/g, '√($1)')
    .replace(/\^\{2\}/g, '²').replace(/\^\{3\}/g, '³').replace(/\^\{([^{}]*)\}/g, (m, t) => sobrescrito(t))
    .replace(/\\cdot/g, '·').replace(/\\times/g, '×').replace(/\\div/g, '÷')
    .replace(/\\leq/g, '≤').replace(/\\geq/g, '≥').replace(/\\neq/g, '≠').replace(/\\pi/g, 'π')
    .replace(/\{,\}/g, ',').replace(/\\,/g, ' ').replace(/\s+/g, ' ').trim();
}
function pôrMat(el, texto){
  el.textContent = texto == null ? '' : String(texto);
  if (!/\\\(/.test(el.textContent)) return;
  if (window.renderMathInElement){
    try { window.renderMathInElement(el, {delimiters: [{left: '\\(', right: '\\)', display: false}], throwOnError: false}); return; } catch(e){}
  }
  el.textContent = texSimples(el.textContent);
}
function matHTML(texto){ const d = document.createElement('span'); pôrMat(d, texto); return d.innerHTML; }

const Som = {
  ctx: null, mestre: null, mudo: false, vol: 0.6, musica: null, timer: null, prox: 0, passo: 0, bpm: 112,
  iniciar(){
    if (this.ctx) { if (this.ctx.state === 'suspended') this.ctx.resume(); return; }
    try {
      this.ctx = new (window.AudioContext || window.webkitAudioContext)();
      this.mestre = this.ctx.createGain(); this.mestre.connect(this.ctx.destination); this.aplicarVol();
    } catch(e){ this.ctx = null; }
  },
  aplicarVol(){ if (this.mestre) this.mestre.gain.value = this.mudo ? 0 : this.vol; },
  nota(f, inicio, dur, tipo, vol){
    if (!this.ctx) return;
    const o = this.ctx.createOscillator(), g = this.ctx.createGain();
    o.type = tipo || 'triangle'; o.frequency.value = f;
    g.gain.setValueAtTime(0.0001, inicio);
    g.gain.exponentialRampToValueAtTime(vol || 0.15, inicio + 0.015);
    g.gain.exponentialRampToValueAtTime(0.0001, inicio + dur);
    o.connect(g); g.connect(this.mestre); o.start(inicio); o.stop(inicio + dur + 0.05);
  },
  ruido(inicio, dur, vol){
    if (!this.ctx) return;
    const n = Math.floor(this.ctx.sampleRate * dur), b = this.ctx.createBuffer(1, n, this.ctx.sampleRate), d = b.getChannelData(0);
    for (let i = 0; i < n; i++) d[i] = (Math.random() * 2 - 1) * (1 - i / n);
    const s = this.ctx.createBufferSource(), g = this.ctx.createGain(), f = this.ctx.createBiquadFilter();
    f.type = 'highpass'; f.frequency.value = 6000; g.gain.value = vol || 0.08;
    s.buffer = b; s.connect(f); f.connect(g); g.connect(this.mestre); s.start(inicio);
  },
  efeito(nome){
    this.iniciar(); if (!this.ctx) return;
    const t = this.ctx.currentTime;
    const seq = {
      entrou: [[784, 0, .12], [1046, .08, .16]],
      resposta: [[880, 0, .06]],
      certo: [[523, 0, .15], [659, .1, .15], [784, .2, .25]],
      tique: [[1200, 0, .04, 'square', .05]],
      gongo: [[196, 0, .9, 'sine', .3], [294, 0, .9, 'sine', .18]],
      sobe: [[660, 0, .1], [880, .08, .14]],
      tambor: [], fanfarra: [[523, 0, .2], [659, .15, .2], [784, .3, .2], [1046, .45, .5], [1318, .6, .7]]
    }[nome] || [];
    seq.forEach(([f, d, dur, tipo, vol]) => this.nota(f, t + d, dur, tipo || 'triangle', vol || .18));
    if (nome === 'tambor') for (let i = 0; i < 18; i++) this.ruido(t + i * 0.09, 0.08, 0.05 + i * 0.006);
  },
  /* Música: "lobby" (arpejo calmo) ou "pergunta" (pulso que acelera no fim) */
  tocar(modo){
    this.parar(); this.iniciar(); if (!this.ctx || !Config.musica) return;
    this.musica = modo; this.passo = 0; this.prox = this.ctx.currentTime + 0.1; this.bpm = modo === 'lobby' ? 108 : 124;
    this.timer = setInterval(() => this.agendar(), 40);
  },
  acelerar(bpm){ this.bpm = bpm; },
  parar(){ clearInterval(this.timer); this.timer = null; this.musica = null; },
  agendar(){
    if (!this.ctx) return;
    const ACORDES = [[261.6, 329.6, 392], [196, 246.9, 293.7], [220, 261.6, 329.6], [174.6, 220, 261.6]];
    while (this.prox < this.ctx.currentTime + 0.15){
      const dur = 60 / this.bpm / 2, p = this.passo, ac = ACORDES[Math.floor(p / 8) % 4];
      if (this.musica === 'lobby'){
        this.nota(ac[p % 3] * 2, this.prox, dur * 1.6, 'triangle', 0.06);
        if (p % 4 === 0) this.nota(ac[0] / 2, this.prox, dur * 3, 'sine', 0.12);
      } else {
        if (p % 2 === 0) this.nota(ac[0] / 2, this.prox, dur * 0.9, 'square', 0.05);
        this.ruido(this.prox, 0.03, p % 4 === 2 ? 0.05 : 0.025);
        if (p % 8 === 4) this.nota(ac[2] * 2, this.prox, dur, 'triangle', 0.05);
      }
      this.prox += dur; this.passo++;
    }
  }
};

function tabuadaAleatoria(){
  const l = [], usados = new Set();
  while (l.length < 10){
    const a = 2 + Math.floor(Math.random() * 8), b = 2 + Math.floor(Math.random() * 8), k = a + 'x' + b;
    if (usados.has(k)) continue; usados.add(k);
    const r = a * b, err = new Set([r]);
    [r + a, r - b, a + b, r + 1, r - 1, r + 10].forEach(x => { if (err.size < 4 && x > 0) err.add(x); });
    const o = [...err].slice(0, 4).map(String), pos = Math.floor(Math.random() * 4);
    [o[0], o[pos]] = [o[pos], o[0]];
    l.push(mc(`Quanto é ${a} × ${b}?`, o, pos, 15, `${a} × ${b} = ${r}`));
  }
  return l;
}

function normalizar(q){
  const n = {tipo: TIPOS[q.tipo] ? q.tipo : 'multipla', t: String(q.t || ''), o: Array.isArray(q.o) ? q.o.map(String) : [],
    c: +q.c || 0, num: q.num == null ? '' : String(q.num), tol: +q.tol || 0, tempo: TEMPOS.includes(+q.tempo) ? +q.tempo : 20,
    pontos: [0, 1000, 2000].includes(+q.pontos) ? +q.pontos : 1000, img: q.img || '', e: String(q.e || '')};
  if (n.tipo === 'vf') n.o = ['Verdadeiro', 'Falso'];
  if (n.tipo === 'enquete') n.pontos = 0;
  return n;
}
/* devolve um texto com o problema, ou '' se a pergunta está pronta para o jogo */
function problema(q){
  if (!q.t.trim()) return 'Falta o enunciado.';
  const alts = q.o.filter(x => x.trim());
  if (q.tipo === 'multipla'){ if (alts.length < 2) return 'Escreva pelo menos 2 alternativas.'; if (!(q.o[q.c] || '').trim()) return 'Marque qual alternativa é a certa.'; }
  if (q.tipo === 'ordenar' && alts.length < 2) return 'Escreva pelo menos 2 etapas.';
  if (q.tipo === 'enquete' && alts.length < 2) return 'Escreva pelo menos 2 opções.';
  if (q.tipo === 'numero' && isNaN(lerNumero(q.num))) return 'Escreva a resposta numérica (ex.: 30, 0,75 ou 3/4).';
  return '';
}
/* lê números como os alunos escrevem: 12 · −3 · 0,75 · 3/4 */
function lerNumero(s){
  s = String(s == null ? '' : s).replace(/\s+/g, '').replace(/[−–]/g, '-');
  if (/^-?\d+([.,]\d+)?$/.test(s)) return parseFloat(s.replace(',', '.'));
  const m = s.match(/^(-?\d+)\/(-?\d+)$/);
  if (m && +m[2] !== 0) return +m[1] / +m[2];
  return NaN;
}

/* Entende os formatos mais comuns de apostila, Word, PDF e ChatGPT:
   1) enunciado · a) b) c) d) alternativas · resposta marcada com *, (x), ✓ ou "(correta)"
   linha "Resposta: C" (ou "Gabarito: C") · gabarito no final ("Gabarito: 1-A, 2-C …")
   questões sem alternativas com "Resposta: 30" viram resposta numérica · V ou F · "Explicação: …" */
const RX_Q = /^\s*(?:quest[aã]o|exerc[ií]cio|q\.?)?\s*(\d{1,3})\s*[).:\-–]\s*(.*)$/i;
const RX_Q_SO = /^\s*(?:quest[aã]o|exerc[ií]cio)\s*(\d{1,3})\s*[).:\-–]?\s*(.*)$/i;
const RX_ALT = /^\s*\(?\s*([a-eA-E])\s*[)\].:\-–]\s+(.*)$/;
const RX_CHK = /^\s*[(\[]\s*([xX✓✔ ]?)\s*[)\]]\s*(.+)$/;
const RX_RESP = /^\s*(?:resposta(?:\s+(?:correta|certa))?|resp\.?|gabarito|alternativa\s+(?:correta|certa)|correta)\s*[:=\-–]\s*(.+)$/i;
const RX_EXPL = /^\s*(?:explica[cç][aã]o|justificativa|resolu[cç][aã]o|coment[aá]rio|solu[cç][aã]o)\s*[:\-–]\s*(.+)$/i;
const RX_PAR = /(\d{1,3})\s*[-–:.)=]?\s*\(?\s*(verdadeiro|falso|[a-eA-E]|V|F)\b\)?/g;

function marcadaCerta(txt){
  let t = String(txt).trim(), certo = false;
  const marcas = [/^\*\s*/, /\s*\*$/, /\s*\((?:x|correta|certa|resposta)\)\s*$/i, /^\((?:x)\)\s*/i, /\s*\[x\]\s*$/i, /\s*[✓✔✅]\s*$/, /^[✓✔✅]\s*/, /\s*(?:<-+|←)\s*(?:correta|certa)?\s*$/i];
  marcas.forEach(rx => { if (rx.test(t)){ certo = true; t = t.replace(rx, ''); } });
  return {txt: t.trim(), certo};
}
function lerResposta(r){
  r = String(r).trim().replace(/[.;]\s*$/, '');
  const m = r.match(/^\(?\s*([a-eA-E])\s*\)?(?:[).:\-–\s]|$)/);
  if (m && !/^(verdadeir|falso|falsa)/i.test(r)) return {letra: m[1].toLowerCase()};
  if (/^(v|verdadeiro|verdadeira|certo|certa)$/i.test(r)) return {vf: 0};
  if (/^(f|falso|falsa|errado|errada)$/i.test(r)) return {vf: 1};
  if (!isNaN(lerNumero(r))) return {num: r.replace(/\s+/g, '')};
  return {texto: r};
}
function comoNoLivro(s){
  if (/\\\(/.test(s)) return s;
  return s
    .replace(/(?<![\w\/.,])(\d{1,4})\/(\d{1,4})(?![\w\/])/g, '\\( \\frac{$1}{$2} \\)')
    .replace(/(?<![\w])([a-zA-Z]|\d+)\^(\d{1,3})(?!\d)/g, '\\( $1^{$2} \\)');
}
function separarAlternativasNaLinha(l){
  const rx = /(^|\s)\(?([a-eA-E])[)\]]\s+/g, pos = [];
  let m; while ((m = rx.exec(l))) pos.push(m.index + m[1].length);
  if (pos.length < 2) return [l];
  const partes = []; if (pos[0] > 0) partes.push(l.slice(0, pos[0]));
  pos.forEach((p, i) => partes.push(l.slice(p, pos[i + 1] || l.length)));
  return partes.map(x => x.trim()).filter(Boolean);
}
function lerQuestoesColadas(texto, op){
  op = Object.assign({tempo: 20, livro: true}, op || {});
  let linhas = String(texto || '').replace(/\r/g, '').replace(/ /g, ' ').replace(/\t/g, ' ').replace(/\*\*|__/g, '').split('\n').map(l => l.replace(/^#+\s*/, '').replace(/\s+$/, ''));
  /* gabarito no final */
  const gab = {};
  for (let i = 0; i < linhas.length; i++){
    if (/^\s*gabarito\b/i.test(linhas[i])){
      const resto = linhas.slice(i).join(' ').replace(/^\s*gabarito\s*[:\-–]?/i, '');
      const pares = [...resto.matchAll(RX_PAR)];
      if (pares.length >= 2 || (pares.length === 1 && /\d\s*[-–:.)=]\s*\(?\s*(?:[a-eA-E]|V|F|verdadeiro|falso)\b/i.test(resto))){ pares.forEach(p => { gab[+p[1]] = p[2]; }); linhas = linhas.slice(0, i); break; }
    }
  }
  /* separa em blocos: por numeração (1. 2) Questão 3) ou, sem numeração, por linha em branco */
  const blocos = [];
  const numerado = linhas.filter(l => RX_Q.test(l) && !RX_ALT.test(l)).length >= 1;
  if (numerado){
    let atual = null;
    linhas.forEach(l => {
      const m = RX_Q_SO.exec(l) || (!RX_ALT.test(l) && !RX_RESP.test(l) ? RX_Q.exec(l) : null);
      if (m && (!atual || atual.linhas.length || atual.inicio.trim())){ atual = {n: +m[1], inicio: m[2] || '', linhas: []}; blocos.push(atual); }
      else if (atual){ if (l.trim()) atual.linhas.push(l); }
    });
  } else {
    let atual = null;
    linhas.forEach(l => { if (!l.trim()){ atual = null; return; } if (!atual){ atual = {n: blocos.length + 1, inicio: l, linhas: []}; blocos.push(atual); } else atual.linhas.push(l); });
  }
  const perguntas = [], avisos = [];
  blocos.forEach((b, bi) => {
    /* alternativas na mesma linha do enunciado: "Quanto é 2 + 2? a) 3 b) 4 c) 5" */
    const partes0 = separarAlternativasNaLinha(b.inicio);
    const comecaAlt = partes0.length > 1 && RX_ALT.test(partes0[0]);
    let enun = partes0.length > 1 ? (comecaAlt ? '' : partes0[0].trim()) : b.inicio.trim(), resp = null, expl = '';
    const extras = partes0.length > 1 ? partes0.slice(comecaAlt ? 0 : 1) : [];
    const alts = [];
    let semLetra = [];
    extras.concat(b.linhas).flatMap(separarAlternativasNaLinha).forEach(l => {
      let m;
      if ((m = RX_EXPL.exec(l))){ expl = m[1].trim(); return; }
      if ((m = RX_RESP.exec(l))){ resp = m[1].trim(); return; }
      if ((m = RX_ALT.exec(l))){ const c = marcadaCerta(m[2]); alts.push({letra: m[1].toLowerCase(), txt: c.txt, certo: c.certo}); return; }
      if ((m = RX_CHK.exec(l))){ const c = marcadaCerta(m[2]); alts.push({letra: String.fromCharCode(97 + alts.length), txt: c.txt, certo: c.certo || !!m[1].trim()}); return; }
      if (!alts.length && !enun){ enun = l.trim(); return; }
      if (alts.length) { alts[alts.length - 1].txt += ' ' + l.trim(); return; }
      semLetra.push(l.trim());
    });
    /* formato antigo: enunciado e alternativas em linhas, a certa com * */
    if (!alts.length && semLetra.length >= 2){ semLetra.forEach((l, k) => { const c = marcadaCerta(l); alts.push({letra: String.fromCharCode(97 + k), txt: c.txt, certo: c.certo}); }); semLetra = []; }
    if (semLetra.length) enun = (enun + ' ' + semLetra.join(' ')).trim();
    if (!resp && gab[b.n] != null) resp = gab[b.n];
    const r = resp ? lerResposta(resp) : {};
    const n = bi + 1;
    let q;
    const ehVF = alts.length === 2 && alts.every(a => /^(v|verdadeiro|verdadeira|f|falso|falsa)\b/i.test(a.txt));
    if (alts.length >= 2 && !ehVF){
      let lista = alts.slice();
      let c = lista.findIndex(a => a.certo);
      if (c < 0 && r.letra) c = lista.findIndex(a => a.letra === r.letra);
      if (c < 0 && r.texto) c = lista.findIndex(a => a.txt.toLowerCase() === r.texto.toLowerCase());
      if (c < 0 && r.num != null) c = lista.findIndex(a => lerNumero(a.txt) === lerNumero(r.num));
      if (lista.length > 4){
        const tirar = lista.map((a, k) => k).filter(k => k !== c).pop();
        avisos.push(`Questão ${n}: tinha ${lista.length} alternativas; o Gamifica Live usa até 4, então tirei a ${lista[tirar].letra.toUpperCase()}).`);
        lista.splice(tirar, 1); if (c > tirar) c--;
        while (lista.length > 4){ const t2 = lista.map((a, k) => k).filter(k => k !== c).pop(); lista.splice(t2, 1); if (c > t2) c--; }
      }
      q = {tipo: 'multipla', t: enun, o: lista.map(a => a.txt).concat(['', '', '', '']).slice(0, 4), c, tempo: op.tempo, pontos: 1000, e: expl};
    } else if (ehVF || r.vf != null || /verdadeiro\s+ou\s+falso|\bv\s+ou\s+f\b|\(\s*v\s*\)\s*ou\s*\(\s*f\s*\)/i.test(enun)){
      let c = r.vf != null ? r.vf : -1;
      if (ehVF){ const k = alts.findIndex(a => a.certo); if (k >= 0) c = /^f/i.test(alts[k].txt) ? 1 : 0; }
      q = {tipo: 'vf', t: enun, c, tempo: op.tempo, pontos: 1000, e: expl};
    } else if (r.num != null){
      q = {tipo: 'numero', t: enun, o: [], num: r.num, tol: 0, tempo: Math.max(op.tempo, 30), pontos: 1000, e: expl};
    } else {
      q = {tipo: 'multipla', t: enun, o: ['', '', '', ''], c: -1, tempo: op.tempo, pontos: 1000, e: expl};
      avisos.push(`Questão ${n}: não encontrei alternativas nem a resposta. Complete no editor.`);
    }
    if (op.livro){ q.t = comoNoLivro(q.t); if (q.o) q.o = q.o.map(x => x ? comoNoLivro(x) : x); if (q.e) q.e = comoNoLivro(q.e); }
    const nq = normalizar(q); if (q.c === -1) nq.c = -1;
    perguntas.push(nq);
  });
  return {perguntas, avisos};
}

function desenharEditor(){
  const q = edQuiz();
  $('#edLista').innerHTML = q.perguntas.map(cardHTML).join('');
  document.querySelectorAll('#edLista .q-card').forEach(c => pôrMat(c.querySelector('.previa'), q.perguntas[+c.dataset.i].t || 'Prévia do enunciado'));
  atualizarResumo();
}
$('#edLista').addEventListener('input', e => {
  const c = e.target.closest('.q-card'); if (!c) return;
  const p = edQuiz().perguntas[+c.dataset.i], campo = e.target.dataset.campo;
  if (campo === 't'){ p.t = e.target.value; pôrMat(c.querySelector('.previa'), p.t || 'Prévia do enunciado'); }
  else if (campo === 'o'){ p.o[+e.target.dataset.k] = e.target.value; }
  else if (campo === 'num' || campo === 'e'){ p[campo] = e.target.value; }
  else if (campo === 'tol'){ p.tol = Math.abs(lerNumero(e.target.value)) || 0; }
  else return;
  agendarSalvar();
});

const CAB_CSV = ['tipo', 'pergunta', 'opcao1', 'opcao2', 'opcao3', 'opcao4', 'correta', 'tempo', 'pontos', 'explicacao'];
function campoCsv(v){ return '"' + String(v == null ? '' : v).replace(/"/g, '""') + '"'; }
function paraCsv(lista){
  const linhas = [CAB_CSV].concat(lista.map(normalizar).map(q => {
    let correta = '';
    if (q.tipo === 'multipla') correta = q.c + 1;
    if (q.tipo === 'vf') correta = q.c === 0 ? 'V' : 'F';
    if (q.tipo === 'numero') correta = q.num + (q.tol ? ' ±' + String(q.tol).replace('.', ',') : '');
    const o = q.tipo === 'vf' ? ['', '', '', ''] : [0, 1, 2, 3].map(k => q.o[k] || '');
    return [q.tipo, q.t].concat(o, [correta, q.tempo, q.pontos, q.e]);
  }));
  return '\ufeff' + linhas.map(l => l.map(campoCsv).join(';')).join('\r\n');
}
function lerCsv(texto){
  texto = texto.replace(/^\ufeff/, '');
  const sep = (texto.split('\n')[0].match(/;/g) || []).length >= (texto.split('\n')[0].match(/,/g) || []).length ? ';' : ',';
  const linhas = []; let linha = [], campo = '', aspas = false;
  for (let i = 0; i < texto.length; i++){
    const ch = texto[i];
    if (aspas){ if (ch === '"'){ if (texto[i + 1] === '"'){ campo += '"'; i++; } else aspas = false; } else campo += ch; }
    else if (ch === '"') aspas = true;
    else if (ch === sep){ linha.push(campo); campo = ''; }
    else if (ch === '\n' || ch === '\r'){ if (ch === '\r' && texto[i + 1] === '\n') i++; linha.push(campo); campo = ''; if (linha.some(x => x.trim())) linhas.push(linha); linha = []; }
    else campo += ch;
  }
  linha.push(campo); if (linha.some(x => x.trim())) linhas.push(linha);
  return linhas;
}
function csvParaPerguntas(texto){
  const linhas = lerCsv(texto); if (!linhas.length) return {ok: [], ruins: []};
  const cab = linhas[0].map(x => x.trim().toLowerCase());
  const temCab = cab.includes('pergunta') || cab.includes('tipo');
  const ok = [], ruins = [];
  linhas.slice(temCab ? 1 : 0).forEach((l, n) => {
    const tipoTxt = (l[0] || '').trim().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
    const mapa = {multipla: 'multipla', 'multipla escolha': 'multipla', vf: 'vf', 'verdadeiro ou falso': 'vf', numero: 'numero', numerica: 'numero', ordenar: 'ordenar', enquete: 'enquete'};
    const tipo = mapa[tipoTxt] || 'multipla';
    const o = [l[2], l[3], l[4], l[5]].map(x => (x || '').trim());
    const corr = (l[6] || '').trim();
    const q = {tipo, t: (l[1] || '').trim(), o, tempo: +l[7] || 20, pontos: l[8] === '' || l[8] == null ? 1000 : +l[8], e: (l[9] || '').trim()};
    if (tipo === 'multipla') q.c = Math.max(0, (parseInt(corr, 10) || 1) - 1);
    if (tipo === 'vf') q.c = /^f/i.test(corr) ? 1 : 0;
    if (tipo === 'numero'){ const m = corr.split('±'); q.num = m[0].trim(); q.tol = m[1] ? Math.abs(lerNumero(m[1])) || 0 : 0; }
    const nq = normalizar(q);
    if (problema(nq)) ruins.push(n + (temCab ? 2 : 1)); else ok.push(nq);
  });
  return {ok, ruins};
}

/* Supabase Realtime (broadcast). Para testes no mesmo navegador: #local (BroadcastChannel). */
const EVENTOS = ['estado', 'entrar', 'resposta', 'oi'];
function criarCanal(codigo, aoMsg){
  const nome = 'gamifica-live-' + codigo;
  if (location.hash.indexOf('local') >= 0 || !window.supabase){
    const bc = new BroadcastChannel(nome);
    bc.onmessage = e => aoMsg(e.data.ev, e.data.p);
    return Promise.resolve({enviar(ev, p){ bc.postMessage({ev, p}); }, fechar(){ bc.close(); }});
  }
  const sb = window.supabase.createClient(window.SUPABASE_URL, window.SUPABASE_KEY, {auth: {persistSession: false}, realtime: {params: {eventsPerSecond: 30}}});
  const ch = sb.channel(nome, {config: {broadcast: {self: false, ack: false}}});
  EVENTOS.forEach(ev => ch.on('broadcast', {event: ev}, m => aoMsg(ev, m.payload)));
  return new Promise((ok, falha) => {
    let feito = false;
    ch.subscribe(st => {
      if (st === 'SUBSCRIBED' && !feito){ feito = true; ok({enviar(ev, p){ ch.send({type: 'broadcast', event: ev, payload: p}); }, fechar(){ sb.removeChannel(ch); }}); }
      else if ((st === 'CHANNEL_ERROR' || st === 'TIMED_OUT') && !feito){ feito = true; falha(st); }
    });
    setTimeout(() => { if (!feito){ feito = true; falha('TEMPO'); } }, 12000);
  });
}

/* Pontos de uma resposta certa: arredondar( P · (1 − t / (2 · T)) )
   P = pontos base da pergunta (1000, 2000 ou 0); t = tempo que o aluno levou; T = tempo total da pergunta.
   Bônus de sequência: a partir do 2º acerto seguido, +100 por acerto da sequência (no máximo +500). */
function pontosDe(P, t, T){ return Math.round(P * (1 - t / (2 * T))); }
function bonusSeq(seq, P){ return P > 0 ? Math.min(500, Math.max(0, seq - 1) * 100) : 0; }

function estado(){

  for (const [pid, j] of Object.entries(H.jog)) p.jog[pid] = [j.nome, j.av];
  if (H.fase === 'leitura' || H.fase === 'pergunta'){
    const q = H.perg[H.qi];
    p.perg = {tipo: q.tipo, pontos: q.pontos, n: q.tipo === 'ordenar' ? H.ordem.length : q.o.filter(x => x.trim()).length};
    if (H.cfg.mostrarTexto) p.perg.t = q.t;
    if (q.tipo === 'ordenar') p.perg.o = H.ordem.map(k => q.o[k]);
    else if (H.cfg.mostrarTexto && q.tipo !== 'numero') p.perg.o = q.o.map((x, k) => [k, x]).filter(v => v[1].trim());
    else if (q.tipo !== 'numero') p.perg.ks = q.o.map((x, k) => k).filter(k => q.o[k].trim());
  }
  if (H.fase === 'leitura') p.restaMs = Math.max(0, H.leituraFim - Date.now());
  if (H.fase === 'pergunta'){ p.restaMs = Math.max(0, H.tempoMs - (Date.now() - H.inicio)); p.resp = Object.keys(H.resp); }
  if (H.fase === 'resultado' || H.fase === 'ranking') p.res = H.res;
  if (H.fase === 'fim') p.fin = H.fin;
  return p;
}

  /* limite de mensagens por aluno (evita travar o telão) */
  const agora = Date.now(), m = H.msgs[p.pid] || (H.msgs[p.pid] = {t: agora, n: 0});
  if (agora - m.t > 1000){ m.t = agora; m.n = 0; }
  if (++m.n > 12) return;
  if (H.removidos.includes(p.pid)){ enviarLogo(); return; }
  const j = H.jog[p.pid];
  if (ev === 'oi'){ if (j) j.visto = agora; return; }
  if (ev === 'entrar'){
    if (j){ j.visto = agora; enviarLogo(); return; }
    if (H.fase === 'fim') return;
    let nome = String(p.nome || '').replace(/\s+/g, ' ').trim().slice(0, 20);
    const nAluno = Object.keys(H.jog).length + 1;
    if (nome.length < 2 || !nomeLimpo(nome)) nome = 'Aluno ' + nAluno;
    const usados = new Set(Object.values(H.jog).map(x => x.nome.toLowerCase()));
    let base = nome, k = 2; while (usados.has(nome.toLowerCase())) nome = base + ' ' + (k++);
    const av = typeof p.av === 'string' && p.av.length <= 8 ? p.av : '🙂';
    H.jog[p.pid] = {nome, av, pts: 0, ac: 0, seq: 0, tTot: 0, visto: agora, resps: {}};
    salvarPartida(true);
    Som.efeito('entrou');
    if (H.fase === 'lobby') desenharBolhas();
    enviarLogo(); return;
  }
  if (ev === 'resposta'){
    if (!j || H.fase !== 'pergunta' || p.qi !== H.qi || H.resp[p.pid]){ enviarLogo(); return; }
    j.visto = agora;
    const t = agora - H.inicio;
    if (t > H.tempoMs + 1500) return;
    const q = H.perg[H.qi];
    let r = p.r;
    if (q.tipo === 'numero'){ r = String(r == null ? '' : r).slice(0, 20); }
    else if (q.tipo === 'ordenar'){ if (!Array.isArray(r) || r.length !== H.ordem.length || new Set(r).size !== r.length || r.some(x => !(x >= 0 && x < H.ordem.length))) return; }
    else { r = +r; if (!(r >= 0 && r <= 3) || !(q.o[r] || '').trim()) return; }
    H.resp[p.pid] = {r, ms: Math.min(t, H.tempoMs)};
    salvarPartida(true);
    Som.efeito('resposta');
    atualizarContador(); enviarLogo();
    const n = ativos().length;
    if (Object.keys(H.resp).length >= Math.max(1, n)) setTimeout(() => { if (H.fase === 'pergunta') encerrarPergunta(); }, 400);
  }
}

function ranking(){ return Object.entries(H.jog).sort((a, b) => b[1].pts - a[1].pts || b[1].ac - a[1].ac || a[1].tTot - b[1].tTot || a[1].nome.localeCompare(b[1].nome)); }
/* corrige uma resposta: true, false ou null (enquete) */
function corrigir(q, r){
  if (q.tipo === 'enquete') return null;
  if (q.tipo === 'numero'){ const v = lerNumero(r), c = lerNumero(q.num); return !isNaN(v) && Math.abs(v - c) <= (q.tol || 0) + 1e-9; }
  if (q.tipo === 'ordenar') return r.every((pos, k) => H.ordem[pos] === k);
  return r === q.c;
}
function encerrarPergunta(){
  if (H.fase !== 'pergunta') return;
  clearInterval(H.tick); Som.parar(); Som.efeito('gongo');
  const q = H.perg[H.qi], cont = [0, 0, 0, 0];
  let certos = 0, errados = 0, somaMs = 0;
  const errosCont = {};
  for (const [pid, j] of Object.entries(H.jog)){
    const r = H.resp[pid];
    if (!r){ if (q.tipo !== 'enquete') j.seq = 0; continue; }
    const ok = corrigir(q, r.r);
    if (q.tipo === 'multipla' || q.tipo === 'vf' || q.tipo === 'enquete') cont[r.r]++;
    somaMs += r.ms;
    if (ok === true){
      j.seq++; j.ac++; j.tTot += r.ms; certos++;
      r.ganho = q.pontos > 0 ? pontosDe(q.pontos, r.ms / 1000, q.tempo) + bonusSeq(j.seq, q.pontos) : 0;
      j.pts += r.ganho; r.ok = 1;
    } else if (ok === false){ j.seq = 0; j.tTot += r.ms; errados++; r.ganho = 0; r.ok = 0; }
    else { r.ganho = 0; r.ok = -1; }
    j.resps[H.qi] = {ok: r.ok, ms: r.ms, ganho: r.ganho, r: r.r};
    if (ok === false && q.tipo !== 'ordenar'){
      const rot = q.tipo === 'numero' ? String(r.r).trim().replace(/\./g, ',').replace('-', '−') : q.o[r.r];
      errosCont[rot] = (errosCont[rot] || 0) + 1;
    }
  }
  const topErros = Object.entries(errosCont).sort((a, b) => b[1] - a[1]).slice(0, 3);
  const nResp = Object.keys(H.resp).length;
  H.hist[H.qi] = {t: q.t, tipo: q.tipo, respostas: nResp, certos, errados, jogadores: Object.keys(H.jog).length, tempoMedio: nResp ? somaMs / nResp / 1000 : 0, erroComum: topErros[0] ? [texSimples(topErros[0][0]), topErros[0][1]] : null};
  const pos = {}; ranking().forEach(([pid], i) => pos[pid] = i + 1);
  const mostraPos = H.cfg.ranking !== 'nenhum';
  const por = {};
  for (const [pid, j] of Object.entries(H.jog)){ const r = H.resp[pid]; por[pid] = [r ? r.ok : -2, r ? r.ganho : 0, j.pts, mostraPos ? pos[pid] : 0, j.seq]; }
  let certaTxt = '';
  if (q.tipo === 'multipla' || q.tipo === 'vf') certaTxt = q.o[q.c];
  if (q.tipo === 'numero') certaTxt = q.num;
  if (q.tipo === 'ordenar') certaTxt = q.o.filter(x => x.trim()).map((x, k) => (k + 1) + ') ' + x).join('  ');
  H.res = {tipo: q.tipo, certaTxt, por, e: q.e};
  H.fase = 'resultado'; enviarEstado(); salvarPartida(true);
  /* tela do professor */
  $('#relogio').style.visibility = 'hidden'; $('#bPular').classList.add('oculto');
  if (q.tipo === 'multipla' || q.tipo === 'vf' || q.tipo === 'enquete'){
    document.querySelectorAll('#pAlts .alternativa[data-k]').forEach(a => { const k = +a.dataset.k; if (q.tipo !== 'enquete') a.classList.add(k === q.c ? 'certa' : 'apagada'); a.insertAdjacentHTML('beforeend', `<span class="cont">${cont[k]}</span>`); });
    const ks = q.o.map((x, k) => k).filter(k => q.o[k].trim()), mx = Math.max(1, ...cont);
    $('#grafico').innerHTML = ks.map(k => `<div style="background:${CORES[k]};height:${26 + cont[k] / mx * 120}px;${q.tipo !== 'enquete' && k !== q.c ? 'opacity:.5;' : ''}">${cont[k]}</div>`).join('');
  } else if (q.tipo === 'numero'){
    $('#pAlts').innerHTML = `<div class="alternativa" style="background:${CORES[3]};justify-content:center">✅ Resposta certa: <span data-mat="${esc(q.num)}"></span></div>`; renderMatEm($('#pAlts'));
    $('#grafico').innerHTML = barras([[certos, CORES[3], 'certas'], [errados, CORES[0], 'erradas'], [Object.keys(H.jog).length - nResp, '#7c7f9e', 'sem resposta']]);
  } else if (q.tipo === 'ordenar'){
    $('#pAlts').innerHTML = '<div class="alternativas" style="grid-template-columns:1fr">' + q.o.filter(x => x.trim()).map((x, k) => `<div class="alternativa certa" style="background:${CORES[3]}"><span>${k + 1}º</span><span data-mat="${esc(x)}"></span></div>`).join('') + '</div>'; renderMatEm($('#pAlts'));
    $('#grafico').innerHTML = barras([[certos, CORES[3], 'na ordem certa'], [errados, CORES[0], 'fora de ordem'], [Object.keys(H.jog).length - nResp, '#7c7f9e', 'sem resposta']]);
  }
  $('#grafico').classList.remove('oculto'); $('#pImg').classList.add('oculto');
  $('#pResp').innerHTML = q.tipo === 'enquete' ? `<b>${nResp}</b>votaram` : `<b>${nResp ? Math.round(certos / nResp * 100) : 0}%</b>acertaram`;

  if (q.tipo === 'numero' && topErros.length) $('#errosComuns').textContent = '🔎 Respostas erradas mais comuns: ' + topErros.map(([v, n]) => v + ' (' + n + (n === 1 ? ' aluno' : ' alunos') + ')').join(' · ');

}

/* ===================== Nuvem: quizzes e relatórios na conta do professor ===================== */
/* Quando o Gamifica Live é aberto no site com login, ele usa a conexão do próprio site (window.parent.GamificaNuvem).
   Cada professor só enxerga os próprios dados (regras de segurança do banco). Sem login ou sem internet,
   tudo continua salvo neste aparelho e sobe para a conta quando possível. */
const Nuvem = {
  sb: null, uid: null, estado: 'local', fila: new Set(), apagar: new Set(), ultimo: {}, timer: null, relatorios: [], msg: '', enviando: false,
  conectar(){
    try {
      const p = window.parent;
      if (p && p !== window && p.GamificaNuvem){ const u = p.GamificaNuvem.usuario(); if (u){ this.sb = p.GamificaNuvem.cliente(); this.uid = u; return true; } }
    } catch(e){}
    this.sb = null; this.uid = null; return false;
  },
  marcarTodos(){ meus.forEach(m => { this.ultimo[m.id] = JSON.stringify([m.titulo, m.perguntas]); }); },
  async iniciar(){
    this.marcarTodos();
    if (!this.conectar()){ this.estado = 'local'; this.mostrar(); return; }
    try { await this.sincronizar(); } catch(e){ this.estado = 'erro'; this.mostrar(); }
    setInterval(() => { if (this.sb && (this.estado === 'erro' || this.fila.size || this.apagar.size)) this.enviar(); }, 60000);
  },
  async sincronizar(){
    this.estado = 'sincronizando'; this.mostrar();
    const {data, error} = await this.sb.from('live_quizzes').select('id,titulo,perguntas,atualizado_em').order('atualizado_em', {ascending: false});
    if (error){ this.estado = 'erro'; this.mostrar(); return; }
    const nuvem = new Map((data || []).map(q => [q.id, q])), juntos = [];
    for (const m of meus){
      if (m.nuvem && nuvem.has(m.nuvem)){
        const c = nuvem.get(m.nuvem), tc = Date.parse(c.atualizado_em) || 0;
        if ((m.atualizado || 0) > tc + 1000) this.fila.add(m.id);                     /* mudou aqui depois: sobe */
        else if (tc > (m.atualizado || 0) + 1000){ m.titulo = c.titulo; m.perguntas = c.perguntas; m.atualizado = tc; m.sinc = tc; }  /* mudou em outro aparelho: desce */
        nuvem.delete(m.nuvem); juntos.push(m);
      } else if (m.nuvem){
        if ((m.atualizado || 0) > (m.sinc || 0)){ delete m.nuvem; this.fila.add(m.id); juntos.push(m); }  /* apagado lá, mas editado aqui: guarda */
      } else { this.fila.add(m.id); juntos.push(m); }                                 /* ainda não subiu */
    }
    for (const c of nuvem.values()){ const t = Date.parse(c.atualizado_em) || Date.now(); juntos.push({id: uid(), nuvem: c.id, titulo: c.titulo, perguntas: c.perguntas, atualizado: t, sinc: t}); }
    meus = juntos; gravarLS(CHAVE_QUIZ, meus); this.marcarTodos();
    if (!meus.some(m => m.id === selecionado) && !PRONTOS.some(p => p.id === selecionado)) selecionado = 'p-exemplo';
    this.sincronizando = true;
    await this.enviar();
    await this.carregarRelatorios();
    this.sincronizando = false;
    if ($('#tInicio').classList.contains('on')) desenharInicio();
    this.mostrar();
  },
  agendar(){
    for (const m of meus){ const s = JSON.stringify([m.titulo, m.perguntas]); if (this.ultimo[m.id] !== s){ this.ultimo[m.id] = s; m.atualizado = Date.now(); this.fila.add(m.id); } }
    gravarLS(CHAVE_QUIZ, meus);
    if (!this.sb) return;
    clearTimeout(this.timer); this.timer = setTimeout(() => this.enviar(), 1500);
  },
  excluir(m){ if (m && m.nuvem && this.sb){ this.apagar.add(m.nuvem); clearTimeout(this.timer); this.timer = setTimeout(() => this.enviar(), 500); } },
  salvarRel(){ if (this.sb){ clearTimeout(this.timer); this.timer = setTimeout(() => this.enviar(), 300); } },
  async enviar(){
    if (!this.sb) return;
    if (this.enviando){ clearTimeout(this.timer); this.timer = setTimeout(() => this.enviar(), 1500); return; }
    this.enviando = true; let falhou = false; this.msg = '';
    try {
      for (const nid of [...this.apagar]){ const {error} = await this.sb.from('live_quizzes').delete().eq('id', nid); if (error) falhou = true; else this.apagar.delete(nid); }
      for (const id of [...this.fila]){
        const m = meus.find(x => x.id === id); if (!m){ this.fila.delete(id); continue; }
        const quando = m.atualizado || Date.now();
        const dados = {titulo: String(m.titulo || 'Meu quiz').slice(0, 120), perguntas: m.perguntas, atualizado_em: new Date(quando).toISOString()};
        let r = null;
        if (m.nuvem){ r = await this.sb.from('live_quizzes').update(dados).eq('id', m.nuvem).select('id'); if (!r.error && (!r.data || !r.data.length)){ delete m.nuvem; r = null; } }
        if (!m.nuvem){ r = await this.sb.from('live_quizzes').insert(dados).select('id').single(); if (!r.error && r.data) m.nuvem = r.data.id; }
        if (r && r.error){ falhou = true; if (/check|violat|large|size/i.test(r.error.message || '')) this.msg = 'O quiz "' + m.titulo + '" ficou grande demais para a nuvem (muitas imagens). Use imagens menores.'; }
        else { m.sinc = quando; if ((m.atualizado || 0) <= quando) this.fila.delete(id); }
      }
      gravarLS(CHAVE_QUIZ, meus);
      const h = lerLS(CHAVE_HIST, []); let mudou = false;
      for (const rel of h){ if (!rel.nuvem){ const {error} = await this.sb.from('live_relatorios').insert({dados: rel}); if (!error){ rel.nuvem = true; mudou = true; } else falhou = true; } }
      if (mudou) gravarLS(CHAVE_HIST, h);
    } catch(e){ falhou = true; }
    this.enviando = false;
    this.estado = falhou ? 'erro' : 'ok'; if (!this.sincronizando) this.mostrar();
  },
  async carregarRelatorios(){
    if (!this.sb) return;
    try {
      const {data, error} = await this.sb.from('live_relatorios').select('dados,criado_em').order('criado_em', {ascending: false}).limit(100);
      if (!error && data) this.relatorios = data.map(x => Object.assign({}, x.dados, {nuvem: true}));
    } catch(e){}
  },
  mostrar(){
    const el = $('#nuvemStatus'); if (!el) return;
    el.textContent = {
      local: '💾 Seus quizzes estão salvos só neste aparelho. Entre no site com o seu login para salvá-los na sua conta e abrir em qualquer computador, tablet ou celular.',
      sincronizando: '☁️ Sincronizando com a sua conta…',
      ok: '☁️ Seus quizzes e relatórios ficam salvos na sua conta. Abra o Gamifica Live em qualquer computador, tablet ou celular.',
      erro: '⚠️ Não consegui falar com a nuvem agora. Tudo continua salvo neste aparelho e vai para a sua conta quando a conexão voltar.'
    }[this.estado] + (this.msg ? ' ' + this.msg : '');
  }
};

let tdAtual = null;
async function abrirTarefa(id){
  mostrar('tTarefa'); $('#tdAlunos').innerHTML = '<p>☁️ Carregando as respostas…</p>'; $('#tdPerguntas').innerHTML = ''; $('#tdAlerta').innerHTML = '';
  try {
    const r1 = await Nuvem.sb.from('live_tarefas').select('*').eq('id', id).single();
    if (r1.error) throw r1.error;
    const r2 = await Nuvem.sb.from('live_tarefa_tentativas').select('nome,turma,pontos,acertos,concluida,iniciada_em,concluida_em,respostas').eq('tarefa_id', id).order('pontos', {ascending: false});
    if (r2.error) throw r2.error;
    tdAtual = {t: r1.data, tent: r2.data || []}; desenharTarefa();
  } catch(e){ $('#tdAlunos').innerHTML = '<p class="dica">' + esc(erroTarefas(e)) + '</p>'; }
}
function rotuloResposta(q, r){
  if (r == null) return '';
  if (q.tipo === 'multipla' || q.tipo === 'enquete') return texSimples(q.o[+r] || '');
  if (q.tipo === 'vf') return +r === 0 ? 'Verdadeiro' : 'Falso';
  if (q.tipo === 'numero') return String(r).replace(/\./g, ',');
  return '';
}
function desenharTarefa(){
  const {t, tent} = tdAtual, perg = t.perguntas, url = linkDe(t.codigo);
  const venceu = t.prazo && new Date(t.prazo) < new Date();
  $('#tdTitulo').textContent = '🔗 ' + t.titulo;
  $('#tdInfo').innerHTML = `Link: <b>${esc(url)}</b> · ${perg.length} questão(ões)${t.prazo ? ' · prazo ' + new Date(t.prazo).toLocaleString('pt-BR', {dateStyle: 'short', timeStyle: 'short'}) : ''} · ${t.ativa && !venceu ? '<span class="tag-on">aberta</span>' : '<span class="tag-off">encerrada</span>'}`;
  $('#tdAcoes').innerHTML = `<button class="btn verde" data-td="copiar">📋 Copiar link</button><a class="btn" style="text-decoration:none;background:#128c4a" target="_blank" rel="noopener" href="https://wa.me/?text=${encodeURIComponent('📝 Quiz "' + t.titulo + '": responda pelo link ' + url)}">💬 WhatsApp</a>` +
    `<button class="btn claro" data-td="ativa">${t.ativa ? '⏹️ Encerrar a tarefa' : '▶️ Reabrir a tarefa'}</button><button class="btn claro" data-td="excluir" style="color:#b91c1c">🗑️ Excluir</button>`;
  const concl = tent.filter(x => x.concluida);
  /* por questão */
  const porQ = perg.map((q, i) => {
    let resp = 0, certos = 0, soma = 0; const erros = {};
    tent.forEach(x => { const r = x.respostas && x.respostas[i]; if (!r) return; resp++; soma += r.ms || 0; if (r.ok === 1) certos++; if (r.ok === 0 && r.r != null){ const rot = rotuloResposta(q, r.r); if (rot) erros[rot] = (erros[rot] || 0) + 1; } });
    const top = Object.entries(erros).sort((a, b) => b[1] - a[1])[0];
    return {t: texSimples(q.t), tipo: q.tipo, resp, certos, pct: resp ? Math.round(certos / resp * 100) : 0, tempo: resp ? soma / resp / 1000 : 0, erro: top ? top[0] + ' (' + top[1] + ')' : ''};
  });
  const dificeis = porQ.map((p, i) => [p, i]).filter(([p]) => p.tipo !== 'enquete' && p.resp && p.pct < 50);
  $('#tdAlerta').innerHTML = !tent.length ? '<p class="dica">Ainda ninguém respondeu. Envie o link para a turma!</p>' :
    (dificeis.length ? `<p class="dica">⚠️ <b>Vale retomar em aula:</b> ${dificeis.map(([p, i]) => 'questão ' + (i + 1) + ' (' + p.pct + '% de acerto)').join(', ')}.</p>` : '<p class="dica">✅ Todas as questões tiveram 50% de acerto ou mais.</p>');
  const nCorr = perg.filter(q => q.tipo !== 'enquete').length;
  $('#tdAlunos').innerHTML = `<p>${concl.length} aluno(s) concluíram${tent.length > concl.length ? ' · ' + (tent.length - concl.length) + ' em andamento ou que pararam no meio' : ''}.</p>` +
    '<table><tr><th>#</th><th>Aluno</th><th>Turma</th><th>Pontos</th><th>Acertos</th><th>Situação</th><th>Data</th></tr>' +
    tent.map((x, i) => `<tr><td>${i + 1}º</td><td>${esc(x.nome)}</td><td>${esc(x.turma)}</td><td>${x.pontos}</td><td>${x.acertos} de ${nCorr}</td><td>${x.concluida ? '✅ concluiu' : '⏳ parou na questão ' + (Object.keys(x.respostas || {}).length + 1)}</td><td>${new Date(x.concluida_em || x.iniciada_em).toLocaleString('pt-BR', {dateStyle: 'short', timeStyle: 'short'})}</td></tr>`).join('') + '</table>';
  $('#tdPerguntas').innerHTML = '<table><tr><th>#</th><th>Questão</th><th>Acerto</th><th>Erro mais comum</th><th>Respostas</th><th>Tempo médio</th></tr>' +
    porQ.map((p, i) => `<tr class="${p.tipo !== 'enquete' && p.resp && p.pct < 50 ? 'dificil' : ''}"><td>${i + 1}</td><td>${esc(p.t)}</td><td>${p.tipo === 'enquete' ? '—' : `<span class="barrinha" style="width:${p.pct}px"></span>${p.pct}%`}</td><td>${esc(p.erro) || '—'}</td><td>${p.resp}</td><td>${p.tempo.toFixed(1).replace('.', ',')} s</td></tr>`).join('') + '</table>';
  tdAtual.porQ = porQ; mostrarAbaTd('alunos');
}

document.addEventListener('keydown', e => {
  if (!['Enter', ' ', 'ArrowRight', 'PageDown'].includes(e.key) || e.altKey || e.ctrlKey || e.metaKey) return;
  if (/input|textarea|select|button/i.test(document.activeElement && document.activeElement.tagName) && e.key !== 'PageDown' && e.key !== 'ArrowRight') return;
  if (!$('#modal').classList.contains('oculto') || !$('#sozinho').classList.contains('oculto')) return;
  const visivel = id => $(id) && !$(id).classList.contains('oculto') && !$(id).disabled && $(id).offsetParent !== null;
  let alvo = null;
  if ($('#tSala').classList.contains('on') && visivel('#bComecar')) alvo = '#bComecar';
  else if ($('#tPergunta').classList.contains('on')) alvo = visivel('#bAvancar') ? '#bAvancar' : null;
  else if ($('#tRanking').classList.contains('on')) alvo = '#bProxima';
  if (alvo){ e.preventDefault(); $(alvo).click(); }
});
